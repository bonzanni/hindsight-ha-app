#include <tunables/global>

profile hindsight flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  file,
  # s6-supervise, start-all.sh, uvicorn and pg0 all run INSIDE this one profile
  # and must signal each other on shutdown (s6 -> SIGTERM start-all.sh -> the
  # API drains + stops pg0). AppArmor mediates a signal at BOTH ends: with a
  # send-only rule the RECEIVER has no receive permission, so the kernel returns
  # EPERM to the sender and SIGTERM never reaches start-all.sh — container stop
  # then hangs until the Supervisor's SIGKILL (exit 137), e.g. every nightly
  # cold backup. Allow both directions (send-only here was the real 137 cause;
  # intra-container signalling has no security value, PID namespaces isolate us).
  signal (send,receive),
  capability fowner,
  capability chown,
  capability dac_override,
  capability setuid,
  capability setgid,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # App: python venv, node, nginx, models
  /app/** rix,
  /opt/hindsight/** rix,
  # CPython 3.11 lives in /usr/local (COPY-ed from upstream; trixie's system
  # python is 3.13). The venv interpreter resolves here, so it must be ix.
  /usr/local/** rix,
  /usr/lib/python3*/** r,

  # Persistent data (pg0 lives here)
  /data/** rwk,

  # Network
  network inet stream,
  network inet dgram,
  network unix stream,

  /proc/** r,
  /sys/** r,
}
